  // Vista secundaria: tabla de TODAS las columnas visibles por permiso.
  // Fase 1 es SOLO LECTURA. Edicion fantasma y guardado quedan fuera de alcance.
  function equipmentRows_cor(){
    const host=$('iadm-cor-equipment-list');if(!host)return;
    if($('iadm-cor-project-title'))$('iadm-cor-project-title').textContent=st.selectedProject?.proyecto||'Proyecto sin nombre';
    if($('iadm-cor-project-meta'))$('iadm-cor-project-meta').textContent=
      (st.selectedProject?.id_proyecto?'PP NS '+st.selectedProject.id_proyecto+' · ':'')+
      st.projectRecordsTotal+' equipo(s) autorizados en el proyecto. Solo lectura · Fase 1.';
    const columns=Array.isArray(st.tableColumns)?st.tableColumns:[];
    // Sin botones ni data-record-id, para no abrir el autoguardado heredado.
    // Sin selectores de edicion multiple, para no invocar PATCH del modulo.
    if(!columns.length){
      host.innerHTML='<div class="iadm-cor-matrix-empty">No hay columnas autorizadas para mostrar.</div>';
    }else if(!st.records.length){
      host.innerHTML='<div class="iadm-cor-matrix-empty">El proyecto no contiene equipos dentro de tu alcance.</div>';
    }else{
      const heads=columns.map(field=>
        '<th scope="col" title="'+esc(field)+'">'+esc(label(field))+'</th>'
      ).join('');
      const rows=st.records.map(record=>
        '<tr data-equipment-id="'+esc(record.id_ins_fl)+'">'+
        columns.map(field=>'<td><span class="iadm-cor-matrix-value">'+
          esc(display(field,record[field]))+'</span></td>').join('')+
        '</tr>'
      ).join('');
      host.innerHTML='<div class="iadm-cor-matrix-scroll" role="region" tabindex="0" '+
        'aria-label="Tabla de equipos del proyecto. Desplazamiento horizontal disponible">'+
        '<table class="iadm-cor-matrix"><thead><tr>'+heads+'</tr></thead><tbody>'+rows+
        '</tbody></table></div>'+
        '<p class="iadm-cor-matrix-caption">'+columns.length+' columnas autorizadas · '+
        st.records.length+' equipo(s) en esta pagina · '+st.projectRecordsTotal+
        ' equipo(s) autorizados en total. Usa Anterior/Siguiente para recorrerlos.</p>';
    }
    pager_cor('iadm-cor-equipment-pagination',st.equipmentOffset,EQUIPMENTS_PAGE_SIZE_COR,st.projectRecordsTotal,'equipos');
    const bulkBar=$('iadm-cor-bulk-bar');if(bulkBar)bulkBar.hidden=true;
  }
