$utf8 = [System.Text.UTF8Encoding]::new($false)
$changed = @{}
$lineEndings = @{}
function Replace-Exact([string]$path,[string]$old,[string]$new) {
    $absolute = Join-Path $REPO $path
    if (-not $changed.ContainsKey($path)) {
        $raw = [System.IO.File]::ReadAllText($absolute,[System.Text.Encoding]::UTF8)
        $lineEndings[$path] = if ($raw.Contains("`r`n")) { "`r`n" } else { "`n" }
        $changed[$path] = $raw.Replace("`r`n","`n")
    }
    $old = $old.Replace("`r`n","`n")
    $new = $new.Replace("`r`n","`n")
    $count = ([regex]::Matches($changed[$path],[regex]::Escape($old))).Count
    if ($count -ne 1) { throw "La ruta $path no coincide con el codigo base ($count coincidencias). No se realizo ninguna escritura." }
    $changed[$path] = $changed[$path].Replace($old,$new)
}
